```
│                                                                                      │
│     localStream.getTracks().forEach(track => {                                      │
│       if (track.readyState !== 'ended' && !producedKinds.has(track.kind)) {        │
│         producedKinds.add(track.kind)                                               │
│         sfuManager.produce(track)                                                   │
│       }                                                                              │
│     })                                                                               │
│   }, [localStream, state.isSendTransportCreated])                                   │
│                                                                                      │

```
